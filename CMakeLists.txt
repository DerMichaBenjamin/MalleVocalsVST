cmake_minimum_required(VERSION 3.22)
project(MalleVocals VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MALLE_BUILD_PLUGIN "Build JUCE plugin" ON)
include(FetchContent)
FetchContent_Declare(stretch GIT_REPOSITORY https://github.com/Signalsmith-Audio/signalsmith-stretch.git GIT_TAG a670068d9aeb64913331d5cc29337b19a457a7df)
FetchContent_MakeAvailable(stretch)
add_library(malle_dsp Source/dsp/Engine.cpp Source/Parameters.cpp)
target_include_directories(malle_dsp PUBLIC Source ${stretch_SOURCE_DIR})
target_link_libraries(malle_dsp PUBLIC signalsmith-stretch)
set_target_properties(malle_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(malle_dsp PRIVATE /W4 /utf-8)
else()
 target_compile_options(malle_dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(malle_tests Tests/DspTests.cpp)
target_link_libraries(malle_tests PRIVATE malle_dsp)
add_test(NAME DSP COMMAND malle_tests)
if(MALLE_BUILD_PLUGIN)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.12 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
 set(FORMATS VST3 Standalone)
 if(APPLE)
  list(APPEND FORMATS AU)
 endif()
 juce_add_plugin(MalleVocals COMPANY_NAME "Rohstoff Audio" BUNDLE_ID "audio.rohstoff.mallevocals" PLUGIN_MANUFACTURER_CODE Rhst PLUGIN_CODE MlVc FORMATS ${FORMATS} PRODUCT_NAME "Malle Vocals" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics)
 juce_generate_juce_header(MalleVocals)
 target_sources(MalleVocals PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
 target_link_libraries(MalleVocals PRIVATE malle_dsp juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
 target_compile_definitions(MalleVocals PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endif()
if(MALLE_BUILD_PLUGIN)
 add_executable(malle_integration Tests/PluginTests.cpp)
 target_include_directories(malle_integration PRIVATE Source ${JUCE_SOURCE_DIR}/modules ${CMAKE_CURRENT_BINARY_DIR}/MalleVocals_artefacts/JuceLibraryCode)
 target_compile_definitions(malle_integration PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(malle_integration PRIVATE MalleVocals malle_dsp)
 add_test(NAME PluginStateAndUI COMMAND malle_integration)
endif()
add_executable(malle_analysis Tests/AnalysisTests.cpp)
target_link_libraries(malle_analysis PRIVATE malle_dsp)
add_test(NAME MeasuredDSP COMMAND malle_analysis)
add_executable(malle_render Tests/RenderVocal.cpp)
target_link_libraries(malle_render PRIVATE malle_dsp)
add_executable(malle_crowd_tests Tests/CrowdTests.cpp)
target_link_libraries(malle_crowd_tests PRIVATE malle_dsp)
add_test(NAME CrowdBankDSP COMMAND malle_crowd_tests)
add_executable(malle_feedback_tests Tests/FeedbackTests.cpp)
target_link_libraries(malle_feedback_tests PRIVATE malle_dsp)
add_test(NAME FeedbackRegression COMMAND malle_feedback_tests)
add_executable(malle_character_tests Tests/CharacterTests.cpp)
target_link_libraries(malle_character_tests PRIVATE malle_dsp)
add_test(NAME CharacterRegression COMMAND malle_character_tests)
